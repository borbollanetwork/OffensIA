# OffensIA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build OffensIA v1 — a private MCP orchestration layer that gives an LLM operator (Kimi/GLM) a governed offensive-security interface with enforced scope, an evidence gate, and an audit ledger, installed via one command.

**Architecture:** A model-agnostic FastMCP server (`core/server.py`) exposes neutral `offensia_*` tools. The risky logic lives in small, unit-tested modules (`scope`, `ledger`, `finding`, `adapters`, `agent_register`); the server wires them thinly. Backend engines (recon = crawl4ai, exec = hexstrike) are referenced as dependencies under `deps/` and reached through adapters that normalize their output. A one-command `install.sh` provisions everything and registers the MCP preset into the user's agent config.

**Tech Stack:** Python 3.11+, `mcp` (FastMCP), `requests`, `pytest`, `pytest` monkeypatch/`responses` for HTTP mocking, Bash for the installer, Docker (crawl4ai).

**Spec:** `docs/specs/2026-09-22-offensia-design.md`

## Global Constraints

- Python 3.11+ only.
- Neutral tool namespace: every model-facing MCP tool is named `offensia_*`; no upstream engine name appears in tool names, docstrings, or `README.md`.
- No vendoring: engine source is never copied into the OffensIA source tree; `deps/` is git-ignored and populated at install time.
- `scope.allow` ships empty; default-deny is enforced in code on every target-bearing call.
- Core is model-agnostic; anything model-specific lives only under `presets/`.
- The installer never overwrites a user's agent config without writing a timestamped backup first.
- Result schema returned by every adapter/tool: `{ok: bool, target: str, action: str, raw: str, summary: str, ledger_ref: str|None, error: str|None}`.
- Kimi model id default: `moonshot/kimi-k3`; GLM model id default: `zhipu/glm-4.6` (both overridable via `OFFENSIA_LLM`; confirm exact ids against the live provider panel).

## Review Focus

- **Target normalization** (`scope.py`, Task 2): URLs with scheme/port/credentials/path and uppercase hosts must reduce to the same bare host before matching, or an authorized target is wrongly refused. Test in Task 2.
- **Malformed/empty scope** (`scope.py`, Task 2): a missing `scope.allow`, or one with only comments/blank lines, must yield default-deny with no exception. Test in Task 2.
- **Engine unreachable** (`adapters`, Task 4): when crawl4ai/hexstrike is down or times out, the adapter must return `{ok: false, error: ...}`, never raise into the MCP layer. Test in Task 4.
- **Agent config merge safety** (`agent_register.py`, Task 7): merging into an existing config that already has `mcpServers` (or is malformed JSON) must preserve existing servers and write a restorable backup, never truncate the file. Test in Task 7.
- **Evidence gate integrity** (`finding.py`, Task 5): whitespace-only/trivial `raw_evidence` is rejected, and a `verify_command` that errors or fails leaves status `unconfirmed` — a finding is never `confirmed` without a reproducing command. Test in Task 5.

```python
# core/scope.py
"""Scope enforcement: default-deny allowlist. This is the primary safety control."""
import re
from datetime import datetime, timezone
from pathlib import Path


def host_of(target: str) -> str:
    t = target.strip()
    t = re.sub(r"^[a-zA-Z][a-zA-Z0-9+.-]*://", "", t)
    t = t.split("/")[0]
    t = t.split("@")[-1]
    t = t.split(":")[0]
    return t.lower().strip()


def load_scope(scope_file: Path) -> list[str]:
    if not Path(scope_file).exists():
        return []
    out = []
    for line in Path(scope_file).read_text(encoding="utf-8").splitlines():
        line = line.split("#", 1)[0].strip().lower()
        if line:
            out.append(line)
    return out


def in_scope(target: str, scope_file: Path) -> bool:
    host = host_of(target)
    if not host:
        return False
    for entry in load_scope(scope_file):
        if entry.startswith("*."):
            base = entry[2:]
            if host == base or host.endswith("." + base):
                return True
        elif host == entry:
            return True
    return False


def add_entry(target: str, authorization_ref: str, scope_file: Path) -> None:
    if not authorization_ref or not authorization_ref.strip():
        raise ValueError("authorization_ref is required to add a scope entry")
    Path(scope_file).parent.mkdir(parents=True, exist_ok=True)
    entry = target.strip().lower()
    stamp = datetime.now(timezone.utc).isoformat()
    with Path(scope_file).open("a", encoding="utf-8") as fh:
        fh.write(f"{entry}  # auth={authorization_ref.strip()} added={stamp}\n")
```

```python
# core/ledger.py
"""Append-only per-engagement ledger. The operator cites this, not memory."""
import json
import uuid
from datetime import datetime, timezone
from pathlib import Path

from core.scope import host_of


def _ledger_file(engagements_dir: Path, target: str) -> Path:
    d = Path(engagements_dir) / (host_of(target) or "unknown")
    d.mkdir(parents=True, exist_ok=True)
    return d / "ledger.jsonl"


def append(engagements_dir: Path, target: str, record: dict) -> str:
    ref = uuid.uuid4().hex[:12]
    full = {
        "ledger_ref": ref,
        "ts": datetime.now(timezone.utc).isoformat(),
        "target": target,
        **record,
    }
    path = _ledger_file(engagements_dir, target)
    with path.open("a", encoding="utf-8") as fh:
        fh.write(json.dumps(full, ensure_ascii=False) + "\n")
    return ref


def tail(engagements_dir: Path, target: str, n: int = 40) -> list[dict]:
    path = _ledger_file(engagements_dir, target)
    if not path.exists():
        return []
    lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
    return [json.loads(x) for x in lines[-n:] if x.strip()]
```

```python
# core/adapters/exec.py
"""Adapter for the deterministic tooling engine. Isolates the backend so the
model-facing interface stays stable and neutral."""
import requests
from core.config import settings
from core.adapters.base import result


def run_command(target: str, command: str, use_cache: bool = True) -> dict:
    s = settings()
    try:
        resp = requests.post(
            f"{s['hexstrike_url']}/api/command",
            json={"command": command, "use_cache": use_cache},
            timeout=s["http_timeout"],
        )
        resp.raise_for_status()
        data = resp.json()
    except Exception as exc:  # noqa: BLE001 — network failure must not reach MCP
        return result(False, target, "exec", error=str(exc))
    raw = str(data.get("stdout", ""))
    return result(
        True, target, "exec", raw=raw,
        summary=f"rc={data.get('return_code')} success={data.get('success')}",
    )
```

```python
# core/finding.py
"""Evidence gate: the anti-hallucination control. A finding needs raw evidence;
it is only 'confirmed' when a deterministic command reproduces it."""
from core.adapters.exec import run_command

MIN_EVIDENCE_CHARS = 16


def evaluate(target, raw_evidence, verify_command="", runner=run_command) -> dict:
    if not raw_evidence or len(raw_evidence.strip()) < MIN_EVIDENCE_CHARS:
        return {
            "accepted": False, "status": "unconfirmed",
            "reason": "raw_evidence missing or trivial", "verification": {},
        }
    status, verification = "unconfirmed", {}
    if verify_command.strip():
        vres = runner(target, verify_command)
        verification = {"command": verify_command, "ok": vres.get("ok"),
                        "raw_tail": str(vres.get("raw", ""))[-2000:]}
        if vres.get("ok"):
            status = "confirmed"
    return {"accepted": True, "status": status,
            "reason": "accepted", "verification": verification}
```

```python
from core import ledger as ledger_mod
from core import finding as finding_mod

from core.adapters.recon import fetch as recon_fetch

PATHS = get_paths()
mcp = FastMCP("offensia")


def _scope_error(target):
    return {"ok": False, "target": target, "action": "scope",
            "error": "OUT_OF_SCOPE", "raw": "", "summary": "", "ledger_ref": None,
            "message": (f"'{scope_mod.host_of(target)}' is not authorized. "
                        "Add it with offensia_scope_add only with written authorization. "
                        "Nothing was executed.")}


@mcp.tool()
def offensia_scope_list() -> dict:
    """List targets currently authorized for testing."""
    return {"ok": True, "entries": scope_mod.load_scope(PATHS.scope_file)}


@mcp.tool()
def offensia_scope_add(target: str, authorization_ref: str) -> dict:
    """Authorize a target. authorization_ref (contract/ticket/bounty/LAB) is required."""
    try:
        scope_mod.add_entry(target, authorization_ref, PATHS.scope_file)
    except ValueError as exc:
        return {"ok": False, "error": "NO_AUTHORIZATION", "message": str(exc)}
    return {"ok": True, "entries": scope_mod.load_scope(PATHS.scope_file)}


@mcp.tool()
def offensia_recon(target: str, mode: str = "md") -> dict:
    """Recon / real-content ingestion for a target. Produces raw fact into the ledger."""
    if not scope_mod.in_scope(target, PATHS.scope_file):
        return _scope_error(target)
    res = recon_fetch(target, mode)
    ref = ledger_mod.append(PATHS.engagements_dir, target,
                            {"kind": "recon", "action": "recon", "mode": mode,
                             "ok": res.get("ok"), "summary": res.get("summary")})
    res["ledger_ref"] = ref
    return res


@mcp.tool()
def offensia_exec(target: str, command: str, use_cache: bool = True) -> dict:
    """Run one deterministic security tool command against a target."""
    if not scope_mod.in_scope(target, PATHS.scope_file):
        return _scope_error(target)
    res = exec_run(target, command, use_cache)
    ref = ledger_mod.append(PATHS.engagements_dir, target,
                            {"kind": "exec", "action": "exec", "command": command,
                             "ok": res.get("ok"), "summary": res.get("summary")})
    res["ledger_ref"] = ref
    return res


@mcp.tool()
def offensia_finding(target: str, title: str, severity: str,
                     raw_evidence: str, verify_command: str = "") -> dict:
    """Register a finding. Rejected without raw_evidence; confirmed only if
    verify_command reproduces it."""
    if not scope_mod.in_scope(target, PATHS.scope_file):
        return _scope_error(target)
    verdict = finding_mod.evaluate(target, raw_evidence, verify_command, runner=exec_run)
    if not verdict["accepted"]:
        return {"ok": False, "target": target, "action": "finding",
                "error": "NO_EVIDENCE", "message": verdict["reason"]}
    ref = ledger_mod.append(PATHS.engagements_dir, target,
                            {"kind": "finding", "action": "finding", "title": title,
                             "severity": severity, "status": verdict["status"],
                             "raw_evidence": raw_evidence[:8000],
                             "verification": verdict["verification"]})
    return {"ok": True, "target": target, "action": "finding",
            "status": verdict["status"], "ledger_ref": ref}


@mcp.tool()
def offensia_ledger(target: str, n: int = 40) -> dict:
    """Read the last n ledger records for a target. Cite these, not memory."""
    return {"ok": True, "records": ledger_mod.tail(PATHS.engagements_dir, target, n)}


def main():
    mcp.run()
```

```python
# core/agent_register.py
"""Detect the user's LLM agent config and merge the OffensIA MCP server into it
non-destructively. This is the risky part of the installer, kept pure and tested."""
import glob
import json
import shutil
from datetime import datetime, timezone
from pathlib import Path


def build_server_entry(base: Path, python: str) -> dict:
    return {
        "command": python,
        "args": ["-m", "core.server"],
        "env": {"OFFENSIA_BASE": str(Path(base).resolve())},
    }


def _timestamp() -> str:
    return datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")


def merge_config(config_path: Path, server_entry: dict, name: str = "offensia"):
    config_path = Path(config_path)
    backup = None
    data = {"mcpServers": {}}
    if config_path.exists():
        backup = config_path.with_suffix(config_path.suffix + f".bak.{_timestamp()}")
        shutil.copy2(config_path, backup)
        try:
            data = json.loads(config_path.read_text(encoding="utf-8"))
            if not isinstance(data, dict):
                data = {"mcpServers": {}}
        except json.JSONDecodeError:
            data = {"mcpServers": {}}
    data.setdefault("mcpServers", {})
    data["mcpServers"][name] = server_entry
    config_path.parent.mkdir(parents=True, exist_ok=True)
    config_path.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
    return backup


def restore_backup(config_path: Path) -> bool:
    config_path = Path(config_path)
    backups = sorted(glob.glob(str(config_path) + ".bak.*"))
    if not backups:
        return False
    shutil.copy2(backups[-1], config_path)
    return True
```

```python
"""OffensIA CLI: scope management, reporting, agent (un)registration."""
import argparse
import sys
from pathlib import Path

from core.config import get_paths
from core import scope as scope_mod
from core import ledger as ledger_mod
from core import agent_register as reg


def render_report(engagements_dir, target) -> str:
    records = ledger_mod.tail(engagements_dir, target, 10000)
    findings = [r for r in records if r.get("kind") == "finding"]
    lines = [f"# OffensIA report — {target}", "",
             f"Total ledger records: {len(records)}",
             f"Findings: {len(findings)}", ""]
    for f in findings:
        lines.append(f"- [{f.get('status')}] {f.get('severity')}: {f.get('title')}")
    return "\n".join(lines) + "\n"


def cmd_scope_add(args) -> int:
    paths = get_paths()
    if not args.auth:
        print("error: --auth <authorization_ref> is required", file=sys.stderr)
        return 2
    scope_mod.add_entry(args.target, args.auth, paths.scope_file)
    print(f"authorized: {args.target}")
    return 0


def cmd_scope_list(args) -> int:
    paths = get_paths()
    for e in scope_mod.load_scope(paths.scope_file):
        print(e)
    return 0


def cmd_report(args) -> int:
    paths = get_paths()
    print(render_report(paths.engagements_dir, args.target))
    return 0


def cmd_register(args) -> int:
    paths = get_paths()
    entry = reg.build_server_entry(paths.base, sys.executable)
    backup = reg.merge_config(Path(args.agent_config), entry)
    print(f"registered OffensIA into {args.agent_config}"
          + (f" (backup: {backup})" if backup else ""))
    return 0


def cmd_uninstall(args) -> int:
    ok = reg.restore_backup(Path(args.agent_config))
    print("restored previous agent config" if ok else "no backup found")
    return 0 if ok else 1


def main(argv=None) -> int:
    argv = sys.argv[1:] if argv is None else argv
    p = argparse.ArgumentParser(prog="offensia")
    sub = p.add_subparsers(dest="cmd", required=True)

    sp = sub.add_parser("scope"); ssub = sp.add_subparsers(dest="scmd", required=True)
    a = ssub.add_parser("add"); a.add_argument("target"); a.add_argument("--auth", default="")
    a.set_defaults(func=cmd_scope_add)
    l = ssub.add_parser("list"); l.set_defaults(func=cmd_scope_list)

    r = sub.add_parser("report"); r.add_argument("target"); r.set_defaults(func=cmd_report)
    rg = sub.add_parser("register"); rg.add_argument("--agent-config", dest="agent_config", required=True)
    rg.set_defaults(func=cmd_register)
    un = sub.add_parser("uninstall"); un.add_argument("--agent-config", dest="agent_config", required=True)
    un.set_defaults(func=cmd_uninstall)

    args = p.parse_args(argv)
    return args.func(args)


if __name__ == "__main__":
    raise SystemExit(main())
```

```bash
      git clone -q --depth 1 "$url" "$BASE/deps/$dir"

  done < "$BASE/deps/engines.txt"
```
